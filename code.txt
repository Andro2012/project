#include <IRremote.h>
#include <Servo.h>
Servo s1;
const byte IR_RECEIVE_PIN = 7;
int gradus = 90;


//итак, мы начали...
void setup()
{
pinMode(1,OUTPUT);//желтый, правый мотор(вперед)
pinMode(2,OUTPUT);//оранж, правый мотор (назад)
pinMode(3,OUTPUT);//желтый, левый мотор(вперед)
pinMode(4,OUTPUT);//оранж, левый мотор(назад)

Serial.begin(1);
IrReceiver.begin(IR_RECEIVE_PIN, ENABLE_LED_FEEDBACK);

s1.attach(8);
}


//список функций:

//вперед
void f(int t)
{
digitalWrite(1,1);
digitalWrite(2,0);
digitalWrite(3,1);
digitalWrite(4,0);
delay(t/16);
}


//вправо
void r(int t)
{
digitalWrite(1,0);
digitalWrite(2,1);
digitalWrite(3,1);
digitalWrite(4,0);
delay(t/16);
}


//влево
void l(int t)
{
digitalWrite(1,1);
digitalWrite(2,0);
digitalWrite(3,0);
digitalWrite(4,1);
delay(t/16);
}


//назад
void b(int t)
{
digitalWrite(1,0);
digitalWrite(2,1);
digitalWrite(3,0);
digitalWrite(4,1);
delay(t/16);
}


//стоп моторы
void stop(int t)
{
digitalWrite(1,0);
digitalWrite(2,0);
digitalWrite(3,0);
digitalWrite(4,0);
delay(t/16);
}

void servotuda(int t)
{
s1.write(0);
delay(t/16);
}

void servoobratno(int t)
{
s1.write(180);
delay(t/16);
}


//основные движения

void loop()
{
if (IrReceiver.decode()) // если данные пришли, то выполняем
{
if (IrReceiver.decodedIRData.decodedRawData == 4049) //вперед
{
f(1);
}
else if (IrReceiver.decodedIRData.decodedRawData == 4050) //вправо
{
r(2);
}

else if (IrReceiver.decodedIRData.decodedRawData == 4052) //влево
{
l(2);
}

else if (IrReceiver.decodedIRData.decodedRawData == 4051) //назад
{
b(1);
}
else if (IrReceiver.decodedIRData.decodedRawData == 4010) //кнопка A
{
stop(1);
}

else if (IrReceiver.decodedIRData.decodedRawData == 4011) //кнопка B
{
stop(1);
}

else if (IrReceiver.decodedIRData.decodedRawData == 4012) //кнопка С
{
 servotuda(1000);
}

else if (IrReceiver.decodedIRData.decodedRawData == 4013) //кнопка D
{
 servoobratno(1000);
}


IrReceiver.resume();
}
}
